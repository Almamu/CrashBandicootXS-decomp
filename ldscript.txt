    /* The IWRAM image: IntrMain, the ARM routines and the initialised
     * IWRAM globals, linked to run at 0x03000000 (IntrMain_Buffer) and
     * stored in ROM right after the data above (0x087E55E4). crt0 copies
     * it from __iwram_lma to 0x03000000 at boot; its length is
     * gIntrTable - IntrMain_Buffer. See docs/decomp_dev.md. */
    iwram 0x03000000 : AT(LOADADDR(ROM) + SIZEOF(ROM))
    {
        build/crashbandicootxs/asm/intr_main.o(.text);
        build/crashbandicootxs/src/engine/util/string_arm.o(.text);

    }
    __iwram_lma = LOADADDR(iwram);

    /* Empty cartridge space (0xFF) from the end of the IWRAM image to the
     * end of the ROM: 8 MB, or the next 8 MB when the data runs past it
     * (new level data, docs/levels.md "Adding a room"). */

    {
        BYTE(0xFF);
        . = ALIGN(0x800000);
    } = 0xFF

    /DISCARD/ :
